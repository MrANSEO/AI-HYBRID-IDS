"""
api_client.py
AI-HYBRID-IDS - Phase 16 (composant partage du dashboard)

Petit client HTTP vers l'API FastAPI (Phase 15). Le dashboard ne touche
JAMAIS directement aux fichiers CSV/SQLite - toujours via l'API, pour
respecter la separation d'architecture (section 33: API independante du
dashboard, mais le dashboard, lui, depend de l'API).
"""

import os
import requests

API_BASE_URL = os.environ.get("IDS_API_URL", "http://localhost:8000")


def api_get(path, params=None):
    """
    Retourne (data, error). error est None si tout s'est bien passe,
    sinon un message lisible a afficher dans le dashboard.
    """
    try:
        resp = requests.get(f"{API_BASE_URL}{path}", params=params, timeout=5)
        resp.raise_for_status()
        return resp.json(), None
    except requests.exceptions.ConnectionError:
        return None, (
            f"Impossible de contacter l'API ({API_BASE_URL}). "
            f"Verifiez qu'uvicorn tourne (voir Phase 15)."
        )
    except requests.exceptions.RequestException as e:
        return None, f"Erreur API: {e}"

"""
auth.py - AI-HYBRID-IDS Phase 16 (section 46)
Authentification minimale du dashboard. Mot de passe stocke dans
.streamlit/secrets.toml (jamais dans le code - section 46).
"""
import streamlit as st

def check_password():
    if st.session_state.get("authenticated", False):
        return True

    st.title("AI-HYBRID-IDS — Connexion")
    pwd = st.text_input("Mot de passe", type="password")
    if st.button("Se connecter"):
        try:
            correct = st.secrets["dashboard_password"]
        except (KeyError, FileNotFoundError):
            st.error("secrets.toml manquant ou mal configure (voir .streamlit/secrets.toml)")
            return False
        if pwd == correct:
            st.session_state["authenticated"] = True
            st.rerun()
        else:
            st.error("Mot de passe incorrect")
    return False

    except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
        return False

import math

def sanitize_nan(obj):
    """
    Remplace recursivement tout float NaN par None, quelle que soit sa
    provenance (colonnes pandas float64 avec valeurs manquantes, ex:
    dest_port absent sur les flows ICMP). Necessaire car df.where(...,None)
    seul est insuffisant: pandas reinjecte NaN dans une colonne float64 des
    qu'on tente d'y stocker un None (le dtype ne peut pas contenir de None).
    """
    if isinstance(obj, float) and math.isnan(obj):
        return None
    if isinstance(obj, dict):
        return {k: sanitize_nan(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [sanitize_nan(v) for v in obj]
    return obj

            destination_ips.add(event["dest_ip"])
        if event.get("dest_port") is not None:
            ports.add(event["dest_port"])

    techniques = []
    seen_techniques = set()
    for attack_type in incident.get("attack_types") or []:
        mapping = MITRE_MAPPINGS.get(str(attack_type).strip().lower())
        if mapping and mapping[0] not in seen_techniques:
            technique_id, name = mapping
            techniques.append({
                "id": technique_id,
                "name": name,
                "basis": f"Type observe: {attack_type}",
            })
            seen_techniques.add(technique_id)

    incident["iocs"] = {
        "source_ips": sorted(source_ips),
        "destination_ips": sorted(destination_ips),
        "destination_ports": sorted(ports),
        "protocols": sorted(protocols),
        "domains": [],
        "urls": [],
        "note": "IOC extraits des evenements observes; ils ne constituent pas une preuve absolue.",
    }
    incident["mitre_techniques"] = techniques
    return incident


def validate_config(config):
    """Refuse les reglages qui rendraient les decisions incoherentes."""
    if not isinstance(config, dict):
        raise HTTPException(status_code=422, detail="La configuration doit etre un objet JSON")

    decision = config.get("decision_engine", {})
    if not isinstance(decision, dict):
        raise HTTPException(status_code=422, detail="decision_engine doit etre un objet")

    weight_keys = [
        "weight_suricata", "weight_ml_confidence", "weight_model_agreement",
        "weight_anomaly", "weight_correlation",
    ]
    try:
        weights = [float(decision[key]) for key in weight_keys]
        thresholds = [
            float(decision["threshold_suspicious"]),
            float(decision["threshold_attack"]),
            float(decision["threshold_critical"]),
        ]
    except (KeyError, TypeError, ValueError):
        raise HTTPException(status_code=422, detail="Poids ou seuils decision_engine invalides")

    if any(not math.isfinite(value) or value < 0 or value > 1 for value in weights):
        raise HTTPException(status_code=422, detail="Chaque poids doit etre compris entre 0 et 1")
    if not math.isclose(sum(weights), 1.0, abs_tol=0.01):
        raise HTTPException(status_code=422, detail="La somme des poids doit etre egale a 1")
    if not all(0 <= value <= 1 for value in thresholds) or not (
        thresholds[0] < thresholds[1] < thresholds[2]
    ):
        raise HTTPException(status_code=422, detail="Les seuils doivent respecter 0 <= suspicious < attack < critical <= 1")

    correlation = config.get("correlation", {})
    incident = config.get("incident_engine", {})
    if not isinstance(correlation, dict) or not isinstance(incident, dict):
        raise HTTPException(status_code=422, detail="Les sections correlation et incident_engine sont invalides")
    try:
        window_seconds = int(correlation["window_seconds"])
        timeout_seconds = int(incident["timeout_seconds"])
    except (KeyError, TypeError, ValueError):
        raise HTTPException(status_code=422, detail="Les durees de correlation et d'incident sont invalides")
    if not 10 <= window_seconds <= 3600 or not 60 <= timeout_seconds <= 86400:
        raise HTTPException(status_code=422, detail="Les durees configurees sont hors limites")
    try:
        scan_port_threshold = int(correlation.get("scan_port_threshold", 10))
        repeated_attack_threshold = int(correlation.get("repeated_attack_threshold", 3))
        flood_packets_threshold = float(correlation.get("flood_packets_per_sec_threshold", 1000))
        flood_bytes_threshold = float(correlation.get("flood_bytes_per_sec_threshold", 1_000_000))
    except (TypeError, ValueError):
        raise HTTPException(status_code=422, detail="Les seuils de correlation sont invalides")
    if not 2 <= scan_port_threshold <= 10000 or not 2 <= repeated_attack_threshold <= 1000:
        raise HTTPException(status_code=422, detail="Les seuils scan/repetition sont hors limites")
    if not 1 <= flood_packets_threshold <= 10_000_000 or not 1 <= flood_bytes_threshold <= 10_000_000_000:
        raise HTTPException(status_code=422, detail="Les seuils flood sont hors limites")

    llm = config.get("llm", {})
    if not isinstance(llm, dict):
        raise HTTPException(status_code=422, detail="La section llm est invalide")
    if not isinstance(llm.get("enabled"), bool) or not isinstance(llm.get("model"), str):
        raise HTTPException(status_code=422, detail="Les parametres LLM enabled/model sont invalides")
    try:
        llm_timeout = int(llm["timeout_seconds"])
        max_events = int(llm["max_events_in_prompt"])
    except (KeyError, TypeError, ValueError):
        raise HTTPException(status_code=422, detail="Les parametres LLM numeriques sont invalides")
    if not 30 <= llm_timeout <= 600 or not 1 <= max_events <= 100:
        raise HTTPException(status_code=422, detail="Les limites LLM sont hors limites")

    logs = config.get("logs", {})
    if not isinstance(logs, dict):
        raise HTTPException(status_code=422, detail="La section logs est invalide")
    try:
        retention_days = int(logs["retention_days"])
    except (KeyError, TypeError, ValueError):
        raise HTTPException(status_code=422, detail="La retention des logs est invalide")
    if not 1 <= retention_days <= 3650:
        raise HTTPException(status_code=422, detail="La retention des logs est hors limites")

@app.get("/api/system_health")
def get_system_health():
    """
    Etat systeme: CPU/RAM/disque, statut des services (Suricata, Ollama,
    notre propre detection engine), derniere activite connue.
    """
    import psutil

    cpu_percent = psutil.cpu_percent(interval=0.5)
    mem = psutil.virtual_memory()
    disk = psutil.disk_usage(PROJECT_ROOT)

    def service_active(name):
        try:
            r = subprocess.run(
                ["systemctl", "is-active", name],
                capture_output=True, text=True, timeout=3,
            )
            return r.stdout.strip() == "active"
        except Exception:
            return None  # inconnu (systemctl absent ou timeout) plutot que faux

    def ollama_reachable():
        try:
            r = subprocess.run(
                ["curl", "-s", "-o", "/dev/null", "-w", "%{http_code}",
                 "--max-time", "2", "http://localhost:11434/api/tags"],
                capture_output=True, text=True,
            )
            return r.stdout.strip() == "200"
        except Exception:
            return False

    last_detection_activity = None
    if os.path.exists(DETECTIONS_CSV_PATH):
        try:
            df_last = pd.read_csv(DETECTIONS_CSV_PATH).tail(1)
            if len(df_last) > 0:
                ts_value = df_last.iloc[0]["timestamp"]
                # Une derniere ligne peut etre tronquee (process tue en plein
                # ecriture, ex: pkill pendant les tests) -> pandas lit alors
                # NaN plutot qu'un timestamp. NaN n'est pas serialisable en
                # JSON strict (deja rencontre sur /api/flows) - on le detecte
                # explicitement ici plutot que de laisser planter la reponse.
                if pd.notna(ts_value):
                    last_detection_activity = str(ts_value)
        except Exception:
            pass

    # Verification REELLE que le process est vivant, pas juste que le fichier
    # PID existe (un fichier perime apres un crash donnerait un faux positif -
    # bug corrige suite a relecture, meme logique que /api/status).
    detection_running = False
    detection_pid = None
    detection_pid = _read_detection_pid()
    detection_running = _is_detection_process_running(detection_pid)

    required_models = [
        os.path.join(PROJECT_ROOT, "models", "supervised", "random_forest_smote.pkl"),
        os.path.join(PROJECT_ROOT, "models", "supervised", "xgboost_smote.pkl"),
        os.path.join(PROJECT_ROOT, "models", "supervised", "lightgbm_classweight.pkl"),
        os.path.join(PROJECT_ROOT, "models", "unsupervised", "isolation_forest.pkl"),
    ]
    models_present = all(os.path.exists(p) for p in required_models)

    return {
        "cpu_percent": cpu_percent,
        "cpu_count_logical": psutil.cpu_count(logical=True),
        "memory": {
            "total_gb": round(mem.total / (1024**3), 2),
            "used_gb": round(mem.used / (1024**3), 2),
            "available_gb": round(mem.available / (1024**3), 2),
            "percent": mem.percent,
        },
        "disk": {
            "total_gb": round(disk.total / (1024**3), 2),
            "used_gb": round(disk.used / (1024**3), 2),
            "free_gb": round(disk.free / (1024**3), 2),
            "percent": disk.percent,
        },
        "services": {
            "suricata": service_active("suricata"),
            "ollama": service_active("ollama"),
        },
        "ollama_api_reachable": ollama_reachable(),
        "detection_engine_running": detection_running,
        "detection_engine_pid": detection_pid,
        "models_present": models_present,
        "last_detection_activity": last_detection_activity,
        "warnings": _compute_health_warnings(
            cpu_percent, mem.percent, disk.percent,
            detection_running, service_active("suricata"), models_present,
        ),
    }


def _compute_health_warnings(cpu_pct, mem_pct, disk_pct, detection_running,
                              suricata_active, models_present,
                              cpu_threshold=90, mem_threshold=90, disk_threshold=90):
    """Signaux simples style section 44 (CPU HIGH, RAM HIGH, SURICATA DOWN...)."""
    warnings = []
    if cpu_pct >= cpu_threshold:
        warnings.append("CPU HIGH")
    if mem_pct >= mem_threshold:
        warnings.append("RAM HIGH")
    if disk_pct >= disk_threshold:
        warnings.append("DISK HIGH")
    if suricata_active is False:
        warnings.append("SURICATA DOWN")
    if not detection_running:
        warnings.append("DETECTION ENGINE DOWN")
    if not models_present:
        warnings.append("MODEL ERROR")
    return warnings

@app.post("/api/start")
def start_detection():
    """
    Lance live_detection.py en arriere-plan.

    LIMITE DE SECURITE EXPLICITE (section 53): live_detection.py necessite
    sudo pour lire /var/log/suricata/eve.json (proprietaire root). Cette API
    tourne volontairement SANS sudo (moindre privilege). Pour que ce endpoint
    fonctionne sans intervention manuelle, une regle sudoers NOPASSWD dediee
    et limitee a ce script precis doit etre configuree par l'administrateur -
    ce n'est PAS fait automatiquement ici, et ne doit jamais l'etre de façon
    generique (risque de privilege escalation majeur si mal configure).
    """
    old_pid = _read_detection_pid()
    if _is_detection_process_running(old_pid):
        return {"status": "already_running", "pid": old_pid}

    if not os.path.isfile(VENV_PYTHON_PATH) or not os.path.isfile(DETECTION_SCRIPT_PATH):
        raise HTTPException(status_code=500, detail="Python virtuel ou moteur de detection introuvable")

    os.makedirs(os.path.join(PROJECT_ROOT, "logs"), exist_ok=True)
    log_path = os.path.join(PROJECT_ROOT, "logs", "live_detection_stdout.log")
    log_file = open(log_path, "a")
    try:
        proc = subprocess.Popen(
            ["sudo", "-n", VENV_PYTHON_PATH, DETECTION_SCRIPT_PATH],
            stdout=log_file,
            stderr=subprocess.STDOUT,
        )
    except OSError as e:
        log_file.close()
        raise HTTPException(status_code=500, detail=f"Echec lancement: {e}")
    finally:
        log_file.close()

    os.makedirs(os.path.dirname(PID_FILE_PATH), exist_ok=True)
    with open(PID_FILE_PATH, "w") as f:
        f.write(str(proc.pid))

    return {
        "status": "started",
        "pid": proc.pid,
        "warning": (
            "Necessite une regle sudoers NOPASSWD prealable pour ce script "
            "precis, sinon 'sudo -n' echouera silencieusement (verifier "
            "/api/status juste apres pour confirmer)."
        ),
    }


@app.post("/api/stop")
def stop_detection():
    """Arrete proprement live_detection.py (SIGINT, meme mecanisme que Ctrl+C)."""
    pid = _read_detection_pid()
    if pid is None:
        raise HTTPException(status_code=404, detail="Aucun processus de detection connu")

    if not _is_detection_process_running(pid):
        try:
            os.remove(PID_FILE_PATH)
        except OSError:
            pass
        raise HTTPException(status_code=404, detail="Processus de detection deja arrete")

    try:
        os.kill(pid, signal.SIGINT)
    except ProcessLookupError:
        os.remove(PID_FILE_PATH)
        raise HTTPException(status_code=404, detail="Processus deja arrete")

    os.remove(PID_FILE_PATH)
    return {"status": "stopped", "pid": pid}

report_generator.py

Genere un rapport en francais pour un incident, via Ollama/Qwen local.

REGLES ABSOLUES RESPECTEES:

def _load_llm_config():
    """Charge les reglages LLM sans rendre le rapport dependant de YAML."""
    defaults = {
        "enabled": True,
        "model": OLLAMA_MODEL,
        "timeout_seconds": OLLAMA_TIMEOUT_SECONDS,

def build_prompt(incident):
    """
    Construit un prompt structure a partir des faits de l'incident
    (dict au format retourne par GET /api/incidents/{id}).
    Le prompt liste explicitement les faits et interdit au modele d'en
    inventer d'autres.

    Les evenements sont tronques a MAX_EVENTS_IN_PROMPT (les plus recents) -
    necessaire car certains incidents accumulent des dizaines d'evenements
    (observe empiriquement: un incident a atteint 80 evenements), ce qui
    ferait exploser la taille du prompt et le temps de traitement CPU.
    """
    llm_config = _load_llm_config()
    max_events = max(1, int(llm_config["max_events_in_prompt"]))
    events = incident.get("events", [])
    total_events = len(events)
    truncated = total_events > max_events
    events_to_show = events[-max_events:] if truncated else events

    events_summary = "\n".join(
        f"  - {e['timestamp']} | {e['event_type']} | {e['src_ip']} -> "
        f"{e.get('dest_ip', 'N/A')}:{e.get('dest_port', 'N/A')} | "
        f"{e['detail']} | statut={e['status']} | risk={e['risk_score']}"
        for e in events_to_show
    )
    if truncated:
        omitted = total_events - max_events
        events_summary += (
            f"\n  - [... {omitted} evenement(s) plus ancien(s) non affiches "
            f"ici, mais comptabilises dans le total ci-dessus ...]"
        )

    prompt = f"""Tu es un assistant qui redige des rapports d'incident de securite en francais, pour un mini-SOC.

REGLES STRICTES A RESPECTER:
1. N'utilise QUE les faits fournis ci-dessous. N'invente RIEN.
2. INTERDICTION ABSOLUE de recopier des listes brutes (IP, signatures) en
   puces une par une - resume-les en une phrase (ex: "14 adresses IP
   distinctes ont ete contactees, principalement vers des services cloud").
   Chaque section doit rester COURTE pour laisser de la place aux suivantes.
3. Structure ta reponse en TROIS sections clairement separees et TOUTES
   presentes: "FAITS OBSERVES" (2-3 phrases resumees, pas de liste brute),
   "INTERPRETATION" (ton analyse possible, en precisant que c'est une
   hypothese, pas une certitude), "RECOMMANDATIONS" (exactement 2 actions,
   une phrase courte chacune, moins de 15 mots).
4. Tu ne dois JAMAIS affirmer avec certitude qu'il s'agit d'une attaque
   confirmee si les donnees ne le confirment pas explicitement (severite/statut).
5. Reste concis: 150-200 mots maximum au total pour les TROIS sections.

DONNEES DE L'INCIDENT (faits etablis par le systeme, pas par toi):
- ID incident: {incident['id']}
- IP source: {incident['source_ip']}
- Nombre d'IP destination distinctes: {len(incident['destination_ips'])}
  (exemples: {', '.join(incident['destination_ips'][:3])}{'...' if len(incident['destination_ips']) > 3 else ''})
- Types/signatures observes ({len(incident['attack_types'])} au total): {'; '.join(incident['attack_types'][:4])}{'; ...' if len(incident['attack_types']) > 4 else ''}
- Severite (la plus haute observee): {incident['severity']}
- Score de risque maximum: {incident['max_risk_score']}
- Statut de l'incident: {incident['status']}
- Nombre total d'evenements: {incident['events_count']}
- Premiere activite: {incident['first_seen']}
- Derniere activite: {incident['last_seen']}

EVENEMENTS DE PREUVE (chronologiques):
{events_summary}

Redige maintenant le rapport en francais, en respectant strictement la structure demandee."""

    return prompt


def generate_incident_report(incident):
    """
    Point d'entree principal. Retourne un dict:
    {"success": bool, "report": str|None, "error": str|None}

    Ne leve JAMAIS d'exception - le systeme doit continuer a fonctionner
    meme si Ollama est indisponible (section 47).
    """
    llm_config = _load_llm_config()
    if not llm_config["enabled"]:
        return {"success": False, "report": None, "error": "Generation LLM desactivee dans la configuration"}
    prompt = build_prompt(incident)

    try:
        response = requests.post(
            OLLAMA_URL,
            json={
                "model": llm_config["model"],
                "prompt": prompt,
                "stream": False,
                "options": {
                    # Plafond DUR sur le nombre de tokens generes - le modele
                    # ignore frequemment la consigne textuelle de longueur
                    # (observe empiriquement: 583 tokens generes pour une
                    # consigne de 150-250 mots). Sans ce parametre, la duree
                    # de generation n'est pas bornee de facon fiable.
                    # Plafond releve a 500 (mesure du 12/09/2026: ~5.4 tokens/s
                    # en generation -> +100 tokens ajoute ~18s, marge
                    # confortable sous le timeout de 320s). Motif: a 400
                    # tokens, le rapport se coupait en plein milieu de la
                    # section RECOMMANDATIONS (phrase inachevee) - observe
                    # empiriquement sur incident INC-20260910-9137f030.
                    "num_predict": 500,
                },
            },
            timeout=float(llm_config["timeout_seconds"]),
        )
        response.raise_for_status()
        data = response.json()
        report_text = data.get("response", "").strip()

        if not report_text:
            return {"success": False, "report": None,
                    "error": "Reponse vide recue d'Ollama"}

        return {"success": True, "report": report_text, "error": None}

    except requests.exceptions.ConnectionError:
        return {
            "success": False, "report": None,
            "error": (
                "Ollama indisponible (service non demarre ou modele non "
                f"telecharge - verifier 'ollama list' contient '{llm_config['model']}'). "
                "Le systeme continue de fonctionner normalement sans rapport LLM."
            ),
        }
    except requests.exceptions.Timeout:
        return {
            "success": False, "report": None,
            "error": f"Timeout ({llm_config['timeout_seconds']}s) - generation trop lente sur ce CPU.",
        }
    except requests.exceptions.RequestException as e:
        return {"success": False, "report": None, "error": f"Erreur Ollama: {e}"}

"""
correlation_engine.py
AI-HYBRID-IDS - Phase 12 (sections 9, 28)

Moteur de correlation temporelle par IP source.
Ne prend AUCUNE decision finale (pas de status/severity/risk_score - ce sera
le role du Decision Engine, Phase 13). Se contente de regrouper le contexte
recent (verdicts ML + alertes Suricata) et de calculer des signaux de
correlation simples et transparents.

Regles respectees (section 9):
- Une alerte Suricata n'est jamais ignoree parce que le ML dit "Normal".
- Une detection ML est prise en compte meme sans alerte Suricata correspondante.
"""

import time
from collections import defaultdict, deque


class CorrelationEngine:
    def __init__(self, window_seconds=60):
        self.window_seconds = window_seconds
        # Un deque d'evenements par IP source, trie par temps d'arrivee
        self._events_by_ip = defaultdict(deque)

    def _prune_old(self, src_ip, now):
        """Supprime les evenements plus vieux que la fenetre glissante pour cette IP."""
        events = self._events_by_ip[src_ip]
        while events and (now - events[0]["_internal_time"]) > self.window_seconds:
            events.popleft()

    def add_flow_verdict(self, src_ip, dest_ip, dest_port, predicted_class,
                          confidence, model_agreement, is_anomaly, anomaly_score,
                          flow_id, timestamp_str, packets=0, bytes_total=0,
                          packets_per_sec=0.0, bytes_per_sec=0.0):
        now = time.time()
        self._events_by_ip[src_ip].append({
            "_internal_time": now,
            "type": "ml_verdict",
            "timestamp": timestamp_str,
            "dest_ip": dest_ip,
            "dest_port": dest_port,
            "predicted_class": predicted_class,
            "confidence": confidence,
            "model_agreement": model_agreement,
            "is_anomaly": is_anomaly,
            "anomaly_score": anomaly_score,
            "flow_id": flow_id,
            "packets": packets or 0,
            "bytes_total": bytes_total or 0,
            "packets_per_sec": packets_per_sec or 0.0,
            "bytes_per_sec": bytes_per_sec or 0.0,
        })
        self._prune_old(src_ip, now)

    def add_alert(self, src_ip, signature, category, severity_text,
                  flow_id, timestamp_str):
        now = time.time()
        self._events_by_ip[src_ip].append({
            "_internal_time": now,
            "type": "suricata_alert",
            "timestamp": timestamp_str,
            "signature": signature,
            "category": category,
            "severity_text": severity_text,
            "flow_id": flow_id,
        })
        self._prune_old(src_ip, now)

    def get_context(self, src_ip):
        """
        Retourne le contexte agrege recent pour une IP source:
        signaux bruts, PAS de decision finale.
        """
        now = time.time()
        self._prune_old(src_ip, now)
        events = list(self._events_by_ip[src_ip])

        ml_events = [e for e in events if e["type"] == "ml_verdict"]
        alert_events = [e for e in events if e["type"] == "suricata_alert"]

        distinct_dest_ports = len(set(e["dest_port"] for e in ml_events))
        distinct_dest_ips = len(set(e["dest_ip"] for e in ml_events))
        attack_predictions = [e for e in ml_events if e["predicted_class"] != "Normal Traffic"]
        anomaly_flags = [e for e in ml_events if e["is_anomaly"]]
        total_packets = sum(e["packets"] for e in ml_events)
        total_bytes = sum(e["bytes_total"] for e in ml_events)
        peak_packets_per_sec = max(
            (e["packets_per_sec"] for e in ml_events), default=0.0
        )
        peak_bytes_per_sec = max(
            (e["bytes_per_sec"] for e in ml_events), default=0.0
        )

        return {
            "src_ip": src_ip,
            "window_seconds": self.window_seconds,
            "total_ml_events": len(ml_events),
            "total_alert_events": len(alert_events),
            "distinct_dest_ports": distinct_dest_ports,
            "distinct_dest_ips": distinct_dest_ips,
            "attack_class_count": len(attack_predictions),
            "attack_classes_seen": sorted(set(e["predicted_class"] for e in attack_predictions)),
            "anomaly_count": len(anomaly_flags),
            "total_packets": total_packets,
            "total_bytes": total_bytes,
            "peak_packets_per_sec": peak_packets_per_sec,
            "peak_bytes_per_sec": peak_bytes_per_sec,
            "alert_signatures": [e["signature"] for e in alert_events],
        }

    def evaluate_signals(self, src_ip,
                          scan_port_threshold=10,
                          repeated_attack_threshold=3,
                          flood_packets_per_sec_threshold=1000,
                          flood_bytes_per_sec_threshold=1_000_000):
        """
        Calcule des signaux de correlation simples, explicites et documentes.
        Ce ne sont PAS des verdicts finaux (pas de status/severity) - juste des
        indices transparents que le Decision Engine (Phase 13) pourra utiliser.
        """
        ctx = self.get_context(src_ip)
        signals = []

        if ctx["distinct_dest_ports"] >= scan_port_threshold:
            signals.append(
                f"POSSIBLE_SCAN: {ctx['distinct_dest_ports']} ports distincts "
                f"contactes en {self.window_seconds}s"
            )

        if (
            ctx["peak_packets_per_sec"] >= flood_packets_per_sec_threshold
            or ctx["peak_bytes_per_sec"] >= flood_bytes_per_sec_threshold
        ):
            signals.append(
                f"POSSIBLE_FLOOD: pic={ctx['peak_packets_per_sec']:.1f} paquets/s, "
                f"{ctx['peak_bytes_per_sec']:.1f} octets/s en {self.window_seconds}s"
            )

        if ctx["attack_class_count"] >= repeated_attack_threshold:
            signals.append(
                f"REPEATED_ML_ATTACK_PREDICTION: {ctx['attack_class_count']} flows "
                f"classes hors Normal en {self.window_seconds}s "
                f"(classes: {', '.join(ctx['attack_classes_seen'])})"
            )

        if ctx["total_alert_events"] > 0 and ctx["attack_class_count"] > 0:
            signals.append(
                f"SURICATA_ML_CONVERGENCE: {ctx['total_alert_events']} alerte(s) Suricata "
                f"ET {ctx['attack_class_count']} prediction(s) ML d'attaque pour la meme IP"
            )
        elif ctx["total_alert_events"] > 0:
            # Rule section 9: une alerte Suricata compte MEME si le ML dit Normal
            signals.append(
                f"SURICATA_ALONE: {ctx['total_alert_events']} alerte(s) Suricata "
                f"sans confirmation ML d'attaque - a ne jamais ignorer"
            )

        return ctx, signals

LOW_SEVERITY_FALLBACK_CATEGORIES = {

DEFAULT_SEVERITY_WEIGHT_UNKNOWN = 0.6  # jamais 0 (Regle 7) mais pas non plus 1.0 par defaut


def compute_suricata_severity_weight(signature_severity_text, category):
    """
    Calcule un poids de gravite [0,1] pour une alerte Suricata, en utilisant
    en priorite le texte metadata.signature_severity, puis un repli par
    categorie pour les alertes moteur interne qui n'ont pas cette metadata.
    """
    if signature_severity_text:
        key = signature_severity_text.strip().lower()
        if key in SEVERITY_TEXT_WEIGHTS:
            return SEVERITY_TEXT_WEIGHTS[key]

    if category:
        key = category.strip().lower()
        if key in LOW_SEVERITY_FALLBACK_CATEGORIES:
            return 0.3

    return DEFAULT_SEVERITY_WEIGHT_UNKNOWN


def compute_ml_confidence_adjusted(predicted_class, raw_confidence):
    """Confiance ML, penalisee sur les classes a faible fiabilite reelle (Phase 8)."""
    if predicted_class == "Normal Traffic":
        return 0.0  # pas de "risque" associe a une prediction Normal
    factor = UNRELIABLE_CLASS_FACTORS.get(predicted_class, 1.0)
    return raw_confidence * factor


def compute_model_agreement_score(model_agreement_str):
    """'3/3' -> 1.0, '2/3' -> 0.5, '1/3' -> 0.0"""
    try:
        num, denom = map(int, model_agreement_str.split("/"))
    except (ValueError, AttributeError):
        return 0.0
    if denom == 0:
        return 0.0
    ratio = num / denom
    if ratio == 1.0:
        return 1.0
    elif ratio > 0.5:
        return 0.5
    return 0.0


def compute_correlation_signal_score(correlation_signals):
    """1.0 si un signal fort (section DECISION_ENGINE.md) est present, sinon 0.0."""
    for signal in correlation_signals:
        if any(signal.startswith(strong) for strong in STRONG_CORRELATION_SIGNALS):
            return 1.0
    return 0.0


def compute_risk_score(predicted_class, ml_confidence, model_agreement_str,
                        is_anomaly, suricata_severity_weight, correlation_signals):
    """
    Calcule le risk_score compose selon la formule documentee (DECISION_ENGINE.md).
    suricata_severity_weight: 0.0 si pas d'alerte, sinon poids [0.3-1.0] selon
    la gravite estimee (voir compute_suricata_severity_weight).
    """
    ml_conf_adj = compute_ml_confidence_adjusted(predicted_class, ml_confidence)
    agreement_score = compute_model_agreement_score(model_agreement_str)
    anomaly_component = 1.0 if is_anomaly else 0.0
    correlation_component = compute_correlation_signal_score(correlation_signals)

    risk_score = (
        WEIGHT_SURICATA * suricata_severity_weight
        + WEIGHT_ML_CONFIDENCE * ml_conf_adj
        + WEIGHT_MODEL_AGREEMENT * agreement_score
        + WEIGHT_ANOMALY * anomaly_component
        + WEIGHT_CORRELATION * correlation_component
    )

    breakdown = {
        "suricata_severity_weight": suricata_severity_weight,
        "ml_confidence_adjusted": round(ml_conf_adj, 4),
        "model_agreement_score": agreement_score,
        "anomaly_component": anomaly_component,
        "correlation_component": correlation_component,
    }

    return round(min(risk_score, 1.0), 4), breakdown


def determine_status(risk_score, has_suricata_alert, model_agreement_str, correlation_signals):
    """
    Applique les seuils + les regles absolues (overrides) documentees dans
    DECISION_ENGINE.md.
    """
    if risk_score >= THRESHOLD_CRITICAL:
        status = "CRITICAL"
    elif risk_score >= THRESHOLD_ATTACK:
        status = "ATTACK"
    elif risk_score >= THRESHOLD_SUSPICIOUS:
        status = "SUSPICIOUS"
    else:
        status = "NORMAL"

    # Regle absolue 1 (section 9, Regle 7): une alerte Suricata (quelle que
    # soit sa gravite estimee) ne peut jamais aboutir a un statut NORMAL -
    # garantit une visibilite minimale meme pour une alerte de faible poids.
    if has_suricata_alert and status == "NORMAL":
        status = "SUSPICIOUS"

    # Regle absolue 2 (section 10): desaccord total sans autre signal fort -> UNKNOWN
    agreement_score = compute_model_agreement_score(model_agreement_str)
    has_strong_correlation = compute_correlation_signal_score(correlation_signals) > 0
    if agreement_score == 0.0 and not has_suricata_alert and not has_strong_correlation:
        status = "UNKNOWN"

    return status


def evaluate(predicted_class, ml_confidence, model_agreement_str, is_anomaly,
             has_suricata_alert, correlation_signals,
             suricata_signature_severity_text=None, suricata_category=None):

    Point d'entree principal du Decision Engine.
    Retourne un dict conforme aux champs minimaux de la section 10.

        compute_suricata_severity_weight(suricata_signature_severity_text, suricata_category)

        predicted_class, ml_confidence, model_agreement_str,
        is_anomaly, suricata_severity_weight, correlation_signals,
    )

        "status": status,
        "risk_score": risk_score,
        "attack_type": predicted_class if predicted_class != "Normal Traffic" else None,

        "suricata_severity_weight": suricata_severity_weight,
        "model_agreement": model_agreement_str,
        "correlation_signals": correlation_signals,
        "risk_score_breakdown": breakdown,
    }

def fix_ownership_for_sudo_user(path):
    """
    Corrige le proprietaire d'un fichier/dossier cree sous sudo (root) pour
    qu'il appartienne a l'utilisateur reel - necessaire pour que l'API

    sudo_user = os.environ.get("SUDO_USER")
    if not sudo_user:
        return
    try:
        pw = pwd.getpwnam(sudo_user)
        os.chown(path, pw.pw_uid, pw.pw_gid)
    except (KeyError, PermissionError, FileNotFoundError):
        pass

def prune_detection_log(retention_days):
    """Supprime les detections plus anciennes que la retention configuree."""
    if not os.path.exists(DETECTIONS_CSV_PATH) or retention_days <= 0:
        return
    cutoff = datetime.now().astimezone() - timedelta(days=retention_days)
    fd, pruned_path = tempfile.mkstemp(dir=LOGS_DIR, text=True)
    os.close(fd)
    try:
        with open(DETECTIONS_CSV_PATH, newline="") as source, open(pruned_path, "w", newline="") as target:
            reader = csv.DictReader(source)
            writer = csv.DictWriter(target, fieldnames=CSV_FIELDS)
            writer.writeheader()
            for row in reader:
                try:
                    keep = datetime.fromisoformat(row["timestamp"]) >= cutoff
                except (KeyError, TypeError, ValueError):
                    keep = True
                if keep:
                    writer.writerow(row)
        os.replace(pruned_path, DETECTIONS_CSV_PATH)
        fix_ownership_for_sudo_user(DETECTIONS_CSV_PATH)
    finally:
        if os.path.exists(pruned_path):
            os.remove(pruned_path)

def predict_flow(features_dict, scaler, label_encoder, models, iso_forest):
    """Scaling -> vote des 3 modeles -> confiance -> score d'anomalie."""
    if list(features_dict.keys()) != FEATURE_ORDER:
        return None

    inference_start = time.perf_counter()
    X = pd.DataFrame([features_dict], columns=FEATURE_ORDER)
    X_scaled = pd.DataFrame(scaler.transform(X), columns=FEATURE_ORDER)

    predictions = {}
    confidences = {}
    for name, model in models.items():
        proba = model.predict_proba(X_scaled)[0]
        pred_idx = int(np.argmax(proba))
        pred_label = label_encoder.inverse_transform([pred_idx])[0]
        predictions[name] = pred_label
        confidences[name] = float(proba[pred_idx])

    votes = list(predictions.values())
    vote_counts = pd.Series(votes).value_counts()
    majority_class = vote_counts.index[0]
    model_agreement = int(vote_counts.iloc[0])

    avg_confidence = np.mean([
        confidences[m] for m in models if predictions[m] == majority_class
    ])

    anomaly_pred = iso_forest.predict(X_scaled)[0]
    anomaly_score = float(iso_forest.decision_function(X_scaled)[0])

    return {
        "individual_predictions": predictions,
        "majority_class": majority_class,
        "model_agreement": f"{model_agreement}/{len(models)}",
        "confidence": round(avg_confidence, 4),
        "is_anomaly": bool(anomaly_pred == -1),
        "anomaly_score": round(anomaly_score, 4),
        "inference_time_ms": round((time.perf_counter() - inference_start) * 1000, 3),
    }


def tail_f(path, heartbeat_every=15, start_at_end=True, follow=True):
    """Lit un fichier en continu ou une seule fois pour traiter un replay."""
    while not os.path.exists(path):
        print(f"En attente de la creation de {path}...")
        time.sleep(2)

    with open(path, "r") as f:
        if start_at_end:
            f.seek(0, os.SEEK_END)
        last_activity = time.time()
        while True:
            line = f.readline()
            if not line:
                if not follow:
                    return
                if time.time() - last_activity > heartbeat_every:
                    print(f"  [toujours a l'ecoute - aucun evenement depuis "
                          f"{heartbeat_every}s, normal si le trafic est calme]")
                    last_activity = time.time()
                time.sleep(0.3)
                continue
            last_activity = time.time()
            yield line

def population_stability_index(reference, current, bins=10):
    """Retourne le PSI d'une feature numerique."""
    reference = np.asarray(reference, dtype=float)
    current = np.asarray(current, dtype=float)
    edges = np.unique(np.quantile(reference, np.linspace(0, 1, bins + 1)))
    if len(edges) < 2:
        return 0.0
    edges[0] = -np.inf
    edges[-1] = np.inf
    ref_counts = np.histogram(reference, bins=edges)[0].astype(float)
    cur_counts = np.histogram(current, bins=edges)[0].astype(float)
    ref_pct = np.maximum(ref_counts / max(ref_counts.sum(), 1), 1e-6)
    cur_pct = np.maximum(cur_counts / max(cur_counts.sum(), 1), 1e-6)
    return float(np.sum((cur_pct - ref_pct) * np.log(cur_pct / ref_pct)))


def compute_drift(project_root=None, limit=5000):
    project_root = project_root or get_project_root()
    train_path = os.path.join(project_root, "data", "processed", "train_classweight.csv")
    live_path = os.path.join(project_root, "logs", "detections.csv")
    scaler_path = os.path.join(project_root, "models", "preprocessing", "scaler.pkl")

    if not all(os.path.exists(path) for path in (train_path, live_path, scaler_path)):
        return {"status": "UNAVAILABLE", "reason": "train, live log ou scaler manquant"}

    train = pd.read_csv(train_path)
    live = pd.read_csv(live_path).tail(limit)
    missing = [feature for feature in FEATURE_ORDER if feature not in live.columns]
    if missing or live.empty:
        return {
            "status": "UNAVAILABLE",
            "reason": "les journaux live ne contiennent pas encore toutes les features",
            "missing_features": missing,
        }

    scaler = joblib.load(scaler_path)
    reference = train[FEATURE_ORDER].dropna()
    current_raw = live[FEATURE_ORDER].apply(pd.to_numeric, errors="coerce").dropna()
    if current_raw.empty:
        return {"status": "UNAVAILABLE", "reason": "aucune feature live numerique exploitable"}

    current = pd.DataFrame(
        scaler.transform(current_raw), columns=FEATURE_ORDER
    )
    results = {}
    for feature in FEATURE_ORDER:
        score = population_stability_index(reference[feature], current[feature])
        results[feature] = {
            "psi": round(score, 6),
            "status": "HIGH" if score >= 0.25 else "MEDIUM" if score >= 0.1 else "LOW",
        }
    high_features = [name for name, result in results.items() if result["status"] == "HIGH"]
    return {
        "status": "DRIFT_HIGH" if high_features else "OK",
        "samples": len(current),
        "features": results,
        "high_drift_features": high_features,
        "recommendation": "Retraining recommended" if high_features else "No action",
    }

extract_features_from_eve.py

import json
import sys
from datetime import datetime

FEATURE_ORDER = [
    "destination_port", "flow_duration", "total_fwd_packets", "total_fwd_bytes",
    "flow_bytes_per_sec", "flow_packets_per_sec", "fwd_packets_per_sec",
    "bwd_packets_per_sec", "avg_packet_size", "fin_flag_present",

]


def parse_suricata_timestamp(ts_str):
    """Parse un timestamp Suricata ISO8601 avec timezone (ex: 2026-09-04T16:31:53.850058+0200)."""
    return datetime.fromisoformat(ts_str)


def extract_features(flow_event):
    """
    Transforme un evenement event_type=flow de Suricata en dict de 12 features.
    Retourne None si l'evenement est invalide/incomplet (pas assez d'info pour extraire).
    """
    if flow_event.get("event_type") != "flow":
        return None

    flow = flow_event.get("flow")
    if flow is None:
        return None

    try:
        start = parse_suricata_timestamp(flow["start"])
        end = parse_suricata_timestamp(flow["end"])
    except (KeyError, ValueError):
        return None

    duration_seconds = (end - start).total_seconds()
    flow_duration_us = int(duration_seconds * 1_000_000)

    pkts_toserver = flow.get("pkts_toserver", 0)
    pkts_toclient = flow.get("pkts_toclient", 0)
    bytes_toserver = flow.get("bytes_toserver", 0)
    bytes_toclient = flow.get("bytes_toclient", 0)

    total_packets = pkts_toserver + pkts_toclient
    total_bytes = bytes_toserver + bytes_toclient

    # Protection division par zero (flows tres courts, duration=0 ou 1 seul paquet)
    if duration_seconds > 0:
        flow_bytes_per_sec = total_bytes / duration_seconds
        flow_packets_per_sec = total_packets / duration_seconds
        fwd_packets_per_sec = pkts_toserver / duration_seconds
        bwd_packets_per_sec = pkts_toclient / duration_seconds
    else:
        # Duree nulle (paquets quasi-simultanes) - eviter Inf, mettre a 0
        # (coherent avec le nettoyage Phase 2 qui a supprime les valeurs Inf/negatives)
        flow_bytes_per_sec = 0.0
        flow_packets_per_sec = 0.0
        fwd_packets_per_sec = 0.0
        bwd_packets_per_sec = 0.0

    avg_packet_size = total_bytes / total_packets if total_packets > 0 else 0.0

    tcp_info = flow_event.get("tcp", {})
    fin_flag = int(tcp_info.get("fin", False))
    psh_flag = int(tcp_info.get("psh", False))
    ack_flag = int(tcp_info.get("ack", False))

    features = {
        "destination_port": flow_event.get("dest_port", 0),
        "flow_duration": flow_duration_us,
        "total_fwd_packets": pkts_toserver,
        "total_fwd_bytes": bytes_toserver,
        "flow_bytes_per_sec": flow_bytes_per_sec,
        "flow_packets_per_sec": flow_packets_per_sec,
        "fwd_packets_per_sec": fwd_packets_per_sec,
        "bwd_packets_per_sec": bwd_packets_per_sec,
        "avg_packet_size": avg_packet_size,
        "fin_flag_present": fin_flag,
        "psh_flag_present": psh_flag,
        "ack_flag_present": ack_flag,
    }

    # Validation section 57 - verifier que le vecteur a exactement les features attendues,
    # dans l'ordre attendu, avant de le considerer valide
    if list(features.keys()) != FEATURE_ORDER:
        print("ERREUR: schema de features incoherent - ne devrait jamais arriver ici",
              file=sys.stderr)
        return None

    return features, flow_event


def main(eve_path, max_flows=10):
    print(f"Lecture de: {eve_path}")
    print(f"Extraction des {max_flows} derniers evenements 'flow' valides\n")

    flow_events = []
    with open(eve_path, "r") as f:
        for line in f:
            try:
                evt = json.loads(line)
            except json.JSONDecodeError:
                continue
            if evt.get("event_type") == "flow":
                flow_events.append(evt)

    flow_events = flow_events[-max_flows:]
    print(f"{len(flow_events)} evenements 'flow' trouves (sur les derniers du fichier)\n")

    for i, evt in enumerate(flow_events):
        result = extract_features(evt)
        print(f"--- Flow #{i+1} ---")
        print(f"  Source: {evt.get('src_ip')}:{evt.get('src_port')} -> "
              f"{evt.get('dest_ip')}:{evt.get('dest_port')} ({evt.get('proto')})")

        if result is None:
            print("  IGNORE - donnees insuffisantes pour extraction\n")
            continue

        features, _ = result
        for name in FEATURE_ORDER:
            print(f"    {name:<22} = {features[name]}")
        print()

"""
incident_engine.py
AI-HYBRID-IDS - Phase 14 (section 29)

Transforme les evenements (verdicts ML + alertes Suricata) en incidents
persistants stockes en SQLite. Un incident regroupe l'activite d'une IP
source sur une fenetre glissante (par defaut 300s de silence = incident
considere termine, passe automatiquement en RESOLVED).

Ne cree/etend un incident que pour les statuts SUSPICIOUS/ATTACK/CRITICAL
(jamais NORMAL, jamais UNKNOWN) - coherent avec le Decision Engine (Phase 13).
Les alertes Suricata brutes generent TOUJOURS au moins un incident (Regle 7),
meme sans confirmation ML, avec un statut plancher SUSPICIOUS.
"""

import json
import sqlite3
import uuid
from datetime import datetime

SEVERITY_RANK = {"SUSPICIOUS": 1, "ATTACK": 2, "CRITICAL": 3}
INCIDENT_TIMEOUT_SECONDS = 300  # 5 min de silence -> incident clos automatiquement


def init_db(conn):
    conn.execute("""
        CREATE TABLE IF NOT EXISTS incidents (
            id TEXT PRIMARY KEY,
            source_ip TEXT NOT NULL,
            first_seen TEXT NOT NULL,
            last_seen TEXT NOT NULL,
            destination_ips TEXT NOT NULL DEFAULT '[]',
            attack_types TEXT NOT NULL DEFAULT '[]',
            status TEXT NOT NULL DEFAULT 'OPEN',
            severity TEXT NOT NULL,
            max_risk_score REAL NOT NULL DEFAULT 0.0,
            events_count INTEGER NOT NULL DEFAULT 0
        )
    """)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS incident_events (
            event_id INTEGER PRIMARY KEY AUTOINCREMENT,
            incident_id TEXT NOT NULL,
            timestamp TEXT,
            event_type TEXT NOT NULL,   -- 'ml_verdict' ou 'suricata_alert'
            src_ip TEXT,
            dest_ip TEXT,
            dest_port INTEGER,
            detail TEXT,                -- classe predite ou signature
            status TEXT,
            risk_score REAL,
            flow_id INTEGER,
            FOREIGN KEY(incident_id) REFERENCES incidents(id)
        )
    """)
    conn.commit()


def _parse_ts(ts_str):
    """Parse un timestamp Suricata ISO8601, avec repli sur l'heure actuelle si absent/invalide."""
    if not ts_str:
        return datetime.now().astimezone()
    try:
        return datetime.fromisoformat(ts_str)
    except ValueError:
        return datetime.now().astimezone()


def _generate_incident_id(event_dt):
    date_str = event_dt.strftime("%Y%m%d")
    return f"INC-{date_str}-{uuid.uuid4().hex[:8]}"


def get_or_create_incident(conn, source_ip, event_timestamp_str, initial_severity):
    """
    Retourne l'ID d'un incident OPEN existant pour cette IP source si son
    dernier evenement date de moins de INCIDENT_TIMEOUT_SECONDS, sinon cree
    un nouvel incident (et cloture l'ancien en RESOLVED s'il existait mais
    etait perime).
    """
    event_dt = _parse_ts(event_timestamp_str)

    cur = conn.execute(
        "SELECT id, last_seen FROM incidents WHERE source_ip=? AND status='OPEN' "
        "ORDER BY last_seen DESC LIMIT 1",
        (source_ip,),
    )
    row = cur.fetchone()

    if row is not None:
        incident_id, last_seen_str = row
        last_seen_dt = _parse_ts(last_seen_str)
        elapsed = abs((event_dt - last_seen_dt).total_seconds())
        if elapsed <= INCIDENT_TIMEOUT_SECONDS:
            return incident_id
        else:
            # Incident perime (silence > timeout) - on le cloture proprement
            conn.execute("UPDATE incidents SET status='RESOLVED' WHERE id=?", (incident_id,))

    new_id = _generate_incident_id(event_dt)
    conn.execute(
        "INSERT INTO incidents (id, source_ip, first_seen, last_seen, "
        "destination_ips, attack_types, status, severity, max_risk_score, events_count) "
        "VALUES (?, ?, ?, ?, '[]', '[]', 'OPEN', ?, 0.0, 0)",
        (new_id, source_ip, event_timestamp_str, event_timestamp_str, initial_severity),
    )
    conn.commit()
    return new_id


def record_event(conn, source_ip, dest_ip, dest_port, detail, event_type,
                  status, risk_score, timestamp, flow_id=None):
    """
    Point d'entree principal (Phase 14). Enregistre un evenement (verdict ML
    ou alerte Suricata) dans l'incident actif pour cette IP source (le cree
    si besoin), et met a jour les champs agreges de l'incident.
    """
    if status not in SEVERITY_RANK:
        return None  # NORMAL ou UNKNOWN - pas d'incident (section 29)

    incident_id = get_or_create_incident(conn, source_ip, timestamp, status)

    cur = conn.execute(
        "SELECT destination_ips, attack_types, severity, max_risk_score, events_count "
        "FROM incidents WHERE id=?",
        (incident_id,),
    )
    dest_ips_json, attack_types_json, current_severity, max_risk, events_count = cur.fetchone()
    dest_ips = set(json.loads(dest_ips_json))
    attack_types = set(json.loads(attack_types_json))

    if dest_ip:
        dest_ips.add(dest_ip)
    if detail and detail != "Normal Traffic":
        attack_types.add(detail)

    new_severity = current_severity
    if SEVERITY_RANK.get(status, 0) > SEVERITY_RANK.get(current_severity, 0):
        new_severity = status

    new_max_risk = max(max_risk, risk_score or 0.0)

    conn.execute(
        "UPDATE incidents SET last_seen=?, destination_ips=?, attack_types=?, "
        "severity=?, max_risk_score=?, events_count=? WHERE id=?",
        (
            timestamp, json.dumps(sorted(dest_ips)), json.dumps(sorted(attack_types)),
            new_severity, new_max_risk, events_count + 1, incident_id,
        ),
    )

    conn.execute(
        "INSERT INTO incident_events (incident_id, timestamp, event_type, src_ip, "
        "dest_ip, dest_port, detail, status, risk_score, flow_id) "
        "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
        (incident_id, timestamp, event_type, source_ip, dest_ip, dest_port,
         detail, status, risk_score, flow_id),
    )
    conn.commit()
    return incident_id

import argparse

def sha256_file(path):
    digest = hashlib.sha256()
    with open(path, "rb") as model_file:
        for chunk in iter(lambda: model_file.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def register_models(project_root, version, status="CANDIDATE"):
    metrics_path = os.path.join(project_root, "results", "metrics", "ml_evaluation_summary.json")
    with open(metrics_path) as metrics_file:
        summary = json.load(metrics_file)

    metric_by_model = {
        item.get("model"): item
        for item in summary.get("comparison_retained_models", [])
    }
    metric_by_model["RandomForest"] = metric_by_model.get("Random Forest", {})
    artifact_paths = {
        "RandomForest": os.path.join(project_root, "models", "supervised", "random_forest_smote.pkl"),
        "XGBoost": os.path.join(project_root, "models", "supervised", "xgboost_smote.pkl"),

        "IsolationForest": os.path.join(project_root, "models", "unsupervised", "isolation_forest.pkl"),
    }
    artifacts = []
    for model_name, path in artifact_paths.items():
        if not os.path.isfile(path):
            continue
        metrics = metric_by_model.get(model_name, {})
        artifacts.append({
            "model": model_name,
            "path": os.path.relpath(path, project_root),
            "sha256": sha256_file(path),
            "macro_f1": metrics.get("f1_macro"),
            "variant": metrics.get("variant"),
        })

    registry_dir = os.path.join(project_root, "models", "registry")
    os.makedirs(registry_dir, exist_ok=True)
    registry_path = os.path.join(registry_dir, "registry.json")
    if os.path.exists(registry_path):
        with open(registry_path) as registry_file:
            registry = json.load(registry_file)
    else:
        registry = {"versions": []}

    entry = {
        "version": version,
        "status": status,
        "registered_at": datetime.now(timezone.utc).isoformat(),
        "dataset": "CICIDS2017-derived processed datasets",
        "feature_contract": "src/features/feature_extraction.py",
        "automatic_deployment": False,
        "artifacts": artifacts,
    }
    registry["versions"] = [item for item in registry["versions"] if item.get("version") != version]

def load_split(path, feature_cols, label_encoder):

def evaluate(model, X_test, y_test, label_encoder, model_name):
    t0 = time.time()
    y_pred = model.predict(X_test)
    inference_time = time.time() - t0

    acc = accuracy_score(y_test, y_pred)
    f1_macro = f1_score(y_test, y_pred, average="macro")
    f1_weighted = f1_score(y_test, y_pred, average="weighted")
    precision, recall, f1_per_class, support = precision_recall_fscore_support(
        y_test, y_pred, labels=range(len(label_encoder.classes_)), zero_division=0
    )
    cm = confusion_matrix(y_test, y_pred, labels=range(len(label_encoder.classes_)))

    print(f"\n--- Evaluation: {model_name} ---")
    print(f"Accuracy:      {acc:.4f}")
    print(f"F1-macro:      {f1_macro:.4f}")
    print(f"F1-weighted:   {f1_weighted:.4f}")
    print(f"Temps inference (test complet): {inference_time:.3f}s "
          f"({inference_time/len(X_test)*1000:.5f} ms/echantillon)")
    print("\nPar classe:")
    report = classification_report(
        y_test, y_pred, target_names=label_encoder.classes_, zero_division=0
    )
    print(report)
    print("Matrice de confusion (lignes=vrai, colonnes=predit):")
    print("Classes:", list(label_encoder.classes_))
    print(cm)

    return {
        "model_name": model_name,
        "accuracy": float(acc),
        "f1_macro": float(f1_macro),
        "f1_weighted": float(f1_weighted),
        "inference_time_total_s": float(inference_time),
        "inference_time_per_sample_ms": float(inference_time / len(X_test) * 1000),
        "per_class": {
            cls: {
                "precision": float(precision[i]),
                "recall": float(recall[i]),
                "f1": float(f1_per_class[i]),
                "support": int(support[i]),
            }
            for i, cls in enumerate(label_encoder.classes_)
        },
        "confusion_matrix": cm.tolist(),

"""
paths_utils.py

Resolution correcte des chemins utilisateur, meme lorsque le script tourne
sous sudo (necessaire pour lire eve.json ou la config Suricata).
"""

import os
import pwd


def get_real_home():
    """
    Resout le VRAI home de l'utilisateur, meme sous sudo.
    os.path.expanduser('~') se resoudrait sinon vers /root quand le script
    est execute via 'sudo python3 script.py', ce qui casse tous les chemins
    relatifs au projet (~/AI-HYBRID-IDS/...).
    """
    sudo_user = os.environ.get("SUDO_USER")
    if sudo_user:
        try:
            return pwd.getpwnam(sudo_user).pw_dir
        except KeyError:
            pass
    return os.path.expanduser("~")


def get_project_root():
    """Retourne la racine du projet, avec surcharge possible au deploiement."""
    configured_root = os.environ.get("IDS_PROJECT_ROOT")
    if configured_root:
        return os.path.abspath(os.path.expanduser(configured_root))

    return os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))

    def test_events_from_same_source_are_grouped(self):

            "2026-09-19T10:00:00+00:00", 1,
        )

            "2026-09-19T10:02:00+00:00", 2,

            "SELECT severity, events_count, max_risk_score FROM incidents WHERE id=?",
            (first_id,),
        ).fetchone()
        self.assertEqual(incident, ("CRITICAL", 2, 0.9))

        first_id = incident_engine.record_event(
            self.connection, "10.0.0.4", "10.0.0.5", 22,
            "Port Scan", "ml_verdict", "SUSPICIOUS", 0.3,
            "2026-09-19T10:00:00+00:00",
        )
        second_id = incident_engine.record_event(

            "2026-09-19T10:06:00+00:00",
        )

        self.assertNotEqual(first_id, second_id)
        old_status = self.connection.execute(
            "SELECT status FROM incidents WHERE id=?", (first_id,)
        ).fetchone()[0]
        self.assertEqual(old_status, "RESOLVED")
